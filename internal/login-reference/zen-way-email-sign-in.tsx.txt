'use client'

import { useEffect, useRef, useState, type FormEvent } from 'react'
import { useRouter } from 'next/navigation'
import { createSupabaseBrowserClient } from '@/lib/supabase-client'

export default function EmailSignIn({ korean }: { korean: boolean }) {
  const router = useRouter()
  const [email, setEmail] = useState('')
  const [token, setToken] = useState('')
  const [sent, setSent] = useState(false)
  const [busy, setBusy] = useState(false)
  const [message, setMessage] = useState('')
  const [cooldown, setCooldown] = useState(0)
  const submitting = useRef(false)
  const enabled = process.env.NEXT_PUBLIC_EMAIL_SIGN_IN_ENABLED === 'true'
  useEffect(() => {
    if (!cooldown) return
    const timer = window.setTimeout(() => setCooldown(value => Math.max(0, value - 1)), 1000)
    return () => window.clearTimeout(timer)
  }, [cooldown])

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (!enabled || submitting.current || (!sent && cooldown > 0)) return
    submitting.current = true
    setBusy(true)
    setMessage('')
    try {
      const supabase = createSupabaseBrowserClient()
      if (sent) {
        const { error } = await supabase.auth.verifyOtp({ email: email.trim(), token: token.trim(), type: 'email' })
        if (error) throw error
        router.replace('/welcome')
        router.refresh()
      } else {
        const { error } = await supabase.auth.signInWithOtp({ email: email.trim(), options: { shouldCreateUser: true } })
        if (error) throw error
        setSent(true)
        setCooldown(60)
        setMessage(korean ? '메일의 인증코드를 입력해 주세요. 스팸함도 확인해 주세요.' : 'Enter the code from your email. Please also check spam.')
      }
    } catch {
      setMessage(sent
        ? (korean ? '코드가 올바르지 않거나 만료됐어요. 다시 확인하거나 새 코드를 요청해 주세요.' : 'The code is invalid or expired. Check it or request a new code.')
        : (korean ? '메일을 보내지 못했어요. 잠시 후 다시 시도하거나 Google·Apple로 로그인해 주세요.' : 'Could not send email. Try later or use Google or Apple.'))
    } finally {
      submitting.current = false
      setBusy(false)
    }
  }
  const field = { width: '100%', minHeight: 48, padding: '10px 12px', border: '1px solid rgba(255,255,255,.8)', background: 'rgba(0,0,0,.35)', color: '#fff', fontSize: 16 } as const
  return <details style={{color:'#fff'}}>
    <summary style={{minHeight:48,display:'flex',alignItems:'center',justifyContent:'center',gap:10,border:'1px solid rgba(255,255,255,.9)',cursor:'pointer',fontSize:15}}><svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" aria-hidden="true"><rect x="3" y="5" width="18" height="14" rx="2"/><path d="m4 7 8 6 8-6"/></svg>{korean ? '이메일로 계속하기' : 'Continue with email'}</summary>
    {!enabled ? <p role="status" style={{fontSize:12,lineHeight:1.7,marginTop:12}}>{korean ? '이메일 로그인은 준비 중입니다. 지금은 Google 또는 Apple로 계속해 주세요.' : 'Email sign-in is coming soon. Please use Google or Apple for now.'}</p> : <form onSubmit={submit} style={{display:'grid',gap:10,paddingTop:14}}>
      <label htmlFor="login-email">{korean ? '이메일' : 'Email'}</label>
      <input id="login-email" type="email" autoComplete="email" required maxLength={254} value={email} disabled={sent || busy} onChange={event=>setEmail(event.target.value)} style={field}/>
      {sent && <><label htmlFor="login-code">{korean ? '인증코드' : 'Verification code'}</label><input id="login-code" inputMode="numeric" autoComplete="one-time-code" pattern="[0-9]{6,10}" minLength={6} maxLength={10} required value={token} disabled={busy} onChange={event=>setToken(event.target.value.replace(/\D/g,''))} style={field}/></>}
      <button type="submit" disabled={busy || (!sent && cooldown>0)} style={{...field,cursor:'pointer'}}>{busy ? '···' : sent ? (korean ? '확인하고 로그인' : 'Verify and sign in') : (korean ? '인증코드 받기' : 'Send code')}</button>
      {sent && <button type="button" disabled={busy || cooldown>0} onClick={()=>{setSent(false);setToken('');setMessage('')}} style={{color:'inherit',fontSize:12,minHeight:44}}>{cooldown>0 ? `${cooldown}s` : (korean ? '다른 이메일 / 코드 다시 요청' : 'Change email / request another code')}</button>}
      <p role="status" aria-live="polite" style={{fontSize:12,lineHeight:1.7}}>{message}</p>
    </form>}
  </details>
}

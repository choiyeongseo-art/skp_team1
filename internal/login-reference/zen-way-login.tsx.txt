function AuthButton({
  type,
  label,
  onClick,
  active = false,
  disabled = false,
  fontFamily,
}: {
  type: 'google' | 'apple' | 'email'
  label: string
  onClick?: () => void
  active?: boolean
  disabled?: boolean
  fontFamily?: string
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      disabled={disabled}
      style={{
        width: '100%',
        height: 50,
        borderRadius: 0,
        border: '1px solid rgba(255,255,255,0.90)',
        background: 'rgba(0,0,0,0.10)',
        color: '#FFFFFF',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        gap: 10,
        fontSize: 15,
        fontWeight: 500,
        lineHeight: 2,
        letterSpacing: '0.05em',
        boxShadow: 'none',
        transform: active ? 'scale(0.985)' : 'scale(1)',
        opacity: disabled && !active ? 0.5 : active ? 0.92 : 1,
        transition: 'transform 140ms ease, opacity 140ms ease, background-color 140ms ease',
        cursor: disabled ? (active ? 'wait' : 'not-allowed') : 'pointer',
      }}
    >
      <ProviderIcon type={type} />
      <span style={fontFamily ? { fontFamily } : undefined}>{label}</span>
    </button>
  )
}

export default function LoginPage() {
  const { uiLang } = useLang()
  const copy = LOGIN_COPY[uiLang]
  const authConfigured = isSupabaseAuthConfigured()
  const appleAuthErrorCopy = APPLE_AUTH_ERROR_COPY[uiLang]
  const [showIntro, setShowIntro] = useState(true)
  const [activeSlide, setActiveSlide] = useState(0)
  const [signingIn, setSigningIn] = useState<'google' | 'apple' | null>(null)
  const [authError, setAuthError] = useState('')
  const [utmContent, setUtmContent] = useState('')
  const router = useRouter()

  useEffect(() => {
    if (!isSupabaseAuthConfigured()) return
    const supabase = createSupabaseBrowserClient()
    supabase.auth.getSession().then(({ data }) => {
      if (data.session) router.replace('/home')
    })
  }, [router])

  // UTM 파라미터 읽기 — 인스타 유입 추적
  useEffect(() => {
    const params = new URLSearchParams(window.location.search)
    const content = params.get('utm_content') ?? ''
    const source = params.get('utm_source') ?? ''
    if (source === 'instagram' && content) {
      setUtmContent(content)
      try { sessionStorage.setItem('zen_utm_content', content) } catch {}
    } else {
      try {
        const stored = sessionStorage.getItem('zen_utm_content')
        if (stored) setUtmContent(stored)
      } catch {}
    }
  }, [])

  useEffect(() => {
    track('login_screen_viewed', {})
  }, [])

  useEffect(() => {
    const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    const introTimer = window.setTimeout(() => {
      setShowIntro(false)
    }, reducedMotion ? 500 : 4000)

    return () => window.clearTimeout(introTimer)
  }, [])

  useEffect(() => {
    if (showIntro) return

    const timer = window.setInterval(() => {
      setActiveSlide((prev) => (prev + 1) % LOGIN_SLIDES.length)
    }, 3750)

    return () => window.clearInterval(timer)
  }, [showIntro])

  async function handleGoogleSignIn() {
    if (signingIn) return
    setAuthError('')
    setSigningIn('google')
    try {
      try{if(localStorage.getItem('zen_analytics_consent')==='granted')sessionStorage.setItem('zen_login_pending',String(Date.now()))}catch{}
      track('login_started',{method:'google',source: utmContent ? `instagram_${utmContent}` : '/login'})
      await startGoogleSignIn()
    } catch {
      track('login_failed',{source:'/login'})
      setSigningIn(null)
      setAuthError(({KR: 'Google 로그인을 완료하지 못했어요. 먼저 둘러보기를 이용할 수 있어요.', EN: 'Google sign-in failed. You can still explore first.', JP: 'Googleログインに失敗しました。まずは閲覧できます。', CN: 'Google 登录失败，您仍可先浏览。', TW: 'Google 登入失敗，您仍可先瀏覽。'})[uiLang])
    }
  }

  async function handleAppleSignIn() {
    if (signingIn) return
    setAuthError('')
    setSigningIn('apple')
    try {
      try{if(localStorage.getItem('zen_analytics_consent')==='granted')sessionStorage.setItem('zen_login_pending',String(Date.now()))}catch{}
      track('login_started',{method:'apple',source: utmContent ? `instagram_${utmContent}` : '/login'})
      await startAppleSignIn()
    } catch (error) {
      track('login_failed',{source:'/login'})
      setSigningIn(null)
      setAuthError(
        error instanceof Error && error.message === 'supabase/not-configured'
          ? appleAuthErrorCopy.unavailable
          : appleAuthErrorCopy.failed,
      )
    }
  }

  return (
    <div
      style={{
        position: 'absolute',
        inset: 0,
        overflowX: 'hidden',
        overflowY: showIntro ? 'hidden' : 'auto',
        backgroundColor: '#080907',
      }}
    >
      {showIntro && (
        <div className="zen-login-intro" aria-hidden="true">
          <h1 className="zen-login-intro__wordmark">ZEN WAY</h1>
        </div>
      )}

      <style>{`
        @keyframes zen-login-intro-backdrop {
          0%, 54% { opacity: 1; }
          100% { opacity: 0; }
        }

        @keyframes zen-login-intro-wordmark {
          0%, 8% {
            opacity: 0;
            filter: blur(0);
            letter-spacing: 0.30em;
            transform: scale(0.985);
          }
          18%, 54% {
            opacity: 1;
            filter: blur(0);
            letter-spacing: 0.30em;
            transform: scale(1);
          }
          100% {
            opacity: 0;
            filter: blur(12px);
            letter-spacing: 0.30em;
            transform: scale(1.08);
          }
        }

        .zen-login-intro {
          position: absolute;
          inset: 0;
          z-index: 120;
          display: flex;
          align-items: center;
          justify-content: center;
          overflow: hidden;
          background: #FFFFFF;
          animation: zen-login-intro-backdrop 4000ms cubic-bezier(0.4, 0, 0.2, 1) forwards;
          pointer-events: none;
          will-change: opacity;
        }

        .zen-login-intro__wordmark {
          margin: 0;
          color: #000000;
          font-family: var(--font-dm-sans);
          font-size: 30px;
          font-weight: 500;
          line-height: 2;
          letter-spacing: 0.30em;
          text-align: center;
          white-space: nowrap;
          animation: zen-login-intro-wordmark 3700ms cubic-bezier(0.4, 0, 0.2, 1) forwards;
          will-change: opacity, filter, letter-spacing, transform;
        }

        @media (prefers-reduced-motion: reduce) {
          .zen-login-intro {
            animation-duration: 500ms;
          }

          .zen-login-intro__wordmark {
            animation-duration: 450ms;
          }
        }
      `}</style>

      <div style={{ position: 'relative', minHeight: '100%', display: 'flex', flexDirection: 'column', justifyContent: 'flex-end', paddingTop: 160 }}>
      {/* ── Full-screen image backdrop ───────────────────── */}
      <div style={{ position: 'absolute', inset: 0, overflow: 'hidden', zIndex: 0 }}>

        {LOGIN_SLIDES.map((slide, index) => (
          <div
            key={slide.src}
            style={{
              position: 'absolute',
              inset: 0,
              opacity: activeSlide === index ? 1 : 0,
              transform: activeSlide === index ? 'scale(1.05)' : 'scale(1.01)',
              transition: 'opacity 1000ms cubic-bezier(0.4,0,0.2,1), transform 3750ms linear',
              willChange: 'opacity, transform',
            }}
          >
            <Image
              src={slide.src}
              alt=""
              fill
              priority={index === 0}
              sizes="390px"
              style={{ objectFit: 'cover', objectPosition: slide.objectPosition }}
            />
            <div
              style={{
                position: 'absolute',
                inset: 0,
                background: slide.overlay,
                pointerEvents: 'none',
              }}
            />
          </div>
        ))}


      </div>

      {/* ── Editorial Content — overlays the full-screen image ─────── */}
      <div style={{ padding: '24px 20px max(24px, env(safe-area-inset-bottom))', position: 'relative', zIndex: 10 }}>

        {/* ZEN WAY — editorial masthead */}
        <h1 style={{
          fontFamily: 'var(--font-dm-sans)',
          fontSize: 30,
          fontWeight: 500,
          lineHeight: 2,
          letterSpacing: '0.30em',
          color: '#FFFFFF',
          textAlign: 'center',
          marginBottom: 15,
          marginTop: 0,
          textShadow: '0 2px 30px rgba(0,0,0,0.90), 0 1px 8px rgba(0,0,0,0.80)',
        }}>
          ZEN WAY
        </h1>

        {/* Title — value proposition */}
        <p style={{
          fontFamily: 'var(--font-bookk-myungjo)',
          fontSize: 18,
          fontWeight: 400,
          lineHeight: 1.6,
          wordBreak: 'keep-all',
          textWrap: 'balance',
          color: '#FFFFFF',
          textAlign: 'center',
          margin: '0 0 20px',
          letterSpacing: '0.05em',
          textShadow: '0 2px 14px rgba(0,0,0,0.20)',
        }}>
          {copy.title}
        </p>

        {/* Auth buttons */}
        <div style={{ display: 'flex', flexDirection: 'column', gap: 10, marginBottom: 18 }}>
          {authConfigured && isGoogleSignInEnabled() && <AuthButton type="google" label={signingIn === 'google' ? '···' : copy.google} active={signingIn === 'google'} disabled={!!signingIn} onClick={handleGoogleSignIn} />}
          {authConfigured && <AuthButton type="apple" label={signingIn === 'apple' ? '···' : copy.apple} active={signingIn === 'apple'} disabled={!!signingIn} onClick={handleAppleSignIn} />}
          {authConfigured && <EmailSignIn korean={uiLang === 'KR'} />}

        </div>

        {authError && (
          <p role="alert" style={{ fontSize: 12.5, lineHeight: 2, color: '#FFFFFF', textAlign: 'center', margin: '-8px 0 12px' }}>
            {authError}
          </p>
        )}

        <div style={{ width: '100%', height: 1, backgroundColor: 'rgba(255,255,255,0.72)', marginBottom: 14 }} />

        {/* Explore first — centered guest action */}
        <Link
          href="/welcome"
          onClick={() => track('guest_started', {source: utmContent ? `instagram_${utmContent}` : '/login'})}
          style={{
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            gap: 8,
            height: 50,
            color: '#FFFFFF',
            textDecoration: 'none',
          }}
        >
          <span style={{ fontSize: 15, fontWeight: 500, lineHeight: 2, letterSpacing: '0.05em' }}>{copy.guest}</span>
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round">
            <path d="M5 12h14M13 6l6 6-6 6" />
          </svg>
        </Link>

        <p style={{ fontSize: 10, lineHeight: 2, color: 'rgba(255,255,255,0.58)', textAlign: 'center', marginTop: 6 }}>
          {({KR: '로그인하면 저장한 장소와 여행 계획이 계정에 보관됩니다.', EN: 'Sign in to keep saved places and plans in your account.', JP: 'ログインするとスポットやプランがアカウントに保存されます。', CN: '登录后，地点和计划保存在账号中。', TW: '登入後，地點和計畫儲存在帳號中。'})[uiLang]}
        </p>
        <nav aria-label="Service documents" style={{display:'flex',justifyContent:'center',gap:16,fontSize:11,lineHeight:2,color:'#FFFFFF',marginTop:14}}>
          <Link href="/privacy" style={{color:'inherit'}}>{({KR:'개인정보처리방침',EN:'Privacy Policy',JP:'プライバシーポリシー',CN:'隐私政策',TW:'隱私權政策'})[uiLang]}</Link>
          <Link href="/terms" style={{color:'inherit'}}>{({KR:'이용약관',EN:'Terms of Service',JP:'利用規約',CN:'服务条款',TW:'服務條款'})[uiLang]}</Link>
        </nav>
      </div>
      </div>

    </div>
  )
}
